cmake_minimum_required(VERSION 3.26)

project(ompnn VERSION 0.1.0 LANGUAGES CXX)

# ompnn: OpenMP port of syclnn.  The same sources build for three targets:
#   OMPNN_TARGET=cpu     host OpenMP only (gcc, clang, nvc++, icpx, ...)
#   OMPNN_TARGET=nvidia  + `omp target` offload to NVIDIA (nvptx) and cuBLAS interop
#   OMPNN_TARGET=amd     + `omp target` offload to AMD (amdgcn) and hipBLAS interop
# and against one CBLAS for the host path: OMPNN_BLAS=openblas | mkl.
set(OMPNN_TARGET "cpu" CACHE STRING "cpu | nvidia | amd")
set(OMPNN_OFFLOAD_ARCH "" CACHE STRING "sm_61;sm_80 (nvidia) or gfx1100 (amd); empty = compiler default")
set(OMPNN_BLAS "openblas" CACHE STRING "openblas | mkl (host CBLAS)")
set(OMPNN_BLAS_ROOT "" CACHE PATH "prefix with lib/libcblas.so + include/cblas.h (OpenBLAS) or the MKL root")
option(OMPNN_BUILD_BENCH "Build bench/train_bench" ON)
option(OMPNN_BUILD_PYTHON "Build the pybind11 module" ON)
option(OMPNN_FAST_MATH "-ffast-math (breaks parity)" OFF)
set(OMPNN_MODULE_NAME "_ompnn" CACHE STRING "Python extension module name")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(OpenMP REQUIRED COMPONENTS CXX)
add_library(ompnn_backend INTERFACE)
target_link_libraries(ompnn_backend INTERFACE OpenMP::OpenMP_CXX)
set(_offload_flags)
set(_offload_link)

# ------------------------------------------------------------------ offload flags per compiler
string(REPLACE ";" "," _archs "${OMPNN_OFFLOAD_ARCH}")
set(_arch_flags)
foreach(_a IN LISTS OMPNN_OFFLOAD_ARCH)
  list(APPEND _arch_flags --offload-arch=${_a})
endforeach()
if(OMPNN_TARGET STREQUAL "nvidia")
  target_compile_definitions(ompnn_backend INTERFACE OMPNN_TARGET_NVIDIA=1)
  find_package(CUDAToolkit REQUIRED)
  get_filename_component(CUDAToolkit_ROOT "${CUDAToolkit_BIN_DIR}" DIRECTORY)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    # one --offload-arch per architecture (a fat binary); CUDA from CUDAToolkit
    set(_offload_flags -fopenmp-targets=nvptx64-nvidia-cuda ${_arch_flags})
    if(CUDAToolkit_ROOT)
      list(APPEND _offload_flags --cuda-path=${CUDAToolkit_ROOT})
    elseif(DEFINED ENV{CUDA_HOME})
      list(APPEND _offload_flags --cuda-path=$ENV{CUDA_HOME})
    endif()
    set(_offload_link ${_offload_flags})
  elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    # gcc's nvptx back end knows sm_30 sm_35 sm_53 sm_70 sm_75 sm_80 only and takes a
    # single ISA: use the highest one not above the lowest requested architecture
    # (PTX for sm_53 runs on Pascal through the driver JIT).
    set(_offload_flags -foffload=nvptx-none)
    if(OMPNN_OFFLOAD_ARCH)
      set(_min 999)
      foreach(_a IN LISTS OMPNN_OFFLOAD_ARCH)
        string(REGEX REPLACE "sm_" "" _n "${_a}")
        if(_n LESS _min)
          set(_min ${_n})
        endif()
      endforeach()
      set(_isa sm_30)
      foreach(_cand 35 53 70 75 80)
        if(NOT _cand GREATER _min)
          set(_isa sm_${_cand})
        endif()
      endforeach()
      list(APPEND _offload_flags "-foffload-options=nvptx-none=-misa=${_isa}")
      message(STATUS "ompnn: gcc nvptx ISA ${_isa} for ${OMPNN_OFFLOAD_ARCH}")
    endif()
    set(_offload_link ${_offload_flags})
  elseif(CMAKE_CXX_COMPILER_ID STREQUAL "NVHPC")
    string(REPLACE "sm_" "cc" _nvarch "${_archs}")
    set(_offload_flags -mp=gpu)
    if(_nvarch)
      list(APPEND _offload_flags -gpu=${_nvarch})
    endif()
    set(_offload_link ${_offload_flags})
  else()
    message(WARNING "ompnn: unknown compiler ${CMAKE_CXX_COMPILER_ID} for NVIDIA offload; add the flags by hand")
  endif()
  target_link_libraries(ompnn_backend INTERFACE CUDA::cublas CUDA::cudart)
  set(OMPNN_GPU_BLAS "cuBLAS ${CUDAToolkit_VERSION}")
elseif(OMPNN_TARGET STREQUAL "amd")
  target_compile_definitions(ompnn_backend INTERFACE OMPNN_TARGET_AMD=1 __HIP_PLATFORM_AMD__=1)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set(_offload_flags -fopenmp-targets=amdgcn-amd-amdhsa ${_arch_flags})
    set(_offload_link ${_offload_flags})
  elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    set(_offload_flags -foffload=amdgcn-amdhsa)
    if(OMPNN_OFFLOAD_ARCH)
      list(GET OMPNN_OFFLOAD_ARCH 0 _gcn_arch) # gcc takes a single -march
      list(APPEND _offload_flags "-foffload-options=amdgcn-amdhsa=-march=${_gcn_arch}")
    endif()
    set(_offload_link ${_offload_flags})
  else()
    message(WARNING "ompnn: unknown compiler ${CMAKE_CXX_COMPILER_ID} for AMD offload; add the flags by hand")
  endif()
  find_package(hip REQUIRED)
  find_package(hipblas REQUIRED)
  target_link_libraries(ompnn_backend INTERFACE hip::host roc::hipblas)
  set(OMPNN_GPU_BLAS "hipBLAS ${hipblas_VERSION}")
elseif(OMPNN_TARGET STREQUAL "cpu")
  target_compile_definitions(ompnn_backend INTERFACE OMPNN_HOST_ONLY=1)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    # gcc compiles every `omp target` region for all installed accel targets by
    # default (gcc-14-offload-nvptx/amdgcn): a host-only build must say so.
    set(_offload_flags -foffload=disable)
    set(_offload_link -foffload=disable)
  elseif(CMAKE_CXX_COMPILER_ID STREQUAL "NVHPC")
    set(_offload_flags -mp=multicore)
    set(_offload_link -mp=multicore)
  endif()
else()
  message(FATAL_ERROR "OMPNN_TARGET must be cpu, nvidia or amd")
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND NOT OMPNN_TARGET STREQUAL "cpu")
  # Ubuntu's default hardening (-fcf-protection=full, -fstack-protector-strong) is
  # rejected by the nvptx/amdgcn offload back ends (ptxas: unknown __stack_chk_guard)
  list(APPEND _offload_flags -fcf-protection=none -fno-stack-protector)
  list(APPEND _offload_link -fcf-protection=none -fno-stack-protector)
endif()
if(_offload_flags)
  target_compile_options(ompnn_backend INTERFACE ${_offload_flags})
  target_link_options(ompnn_backend INTERFACE ${_offload_link})
endif()
# OpenMP::OpenMP_CXX links libomp/libgomp by path; the driver flag on the link
# line is what makes clang add libomptarget and gcc its offload plugins
target_link_options(ompnn_backend INTERFACE ${OpenMP_CXX_FLAGS})
if(OMPNN_FAST_MATH)
  target_compile_options(ompnn_backend INTERFACE -ffast-math)
endif()

# ------------------------------------------------------------------ host CBLAS
if(OMPNN_BLAS STREQUAL "openblas")
  find_path(OMPNN_CBLAS_INCLUDE cblas.h HINTS ${OMPNN_BLAS_ROOT}/include ENV OPENBLAS_ROOT
    PATH_SUFFIXES openblas openblas-openmp openblas-pthread x86_64-linux-gnu/openblas-openmp x86_64-linux-gnu/openblas-pthread)
  find_library(OMPNN_CBLAS_LIB NAMES openblas cblas HINTS ${OMPNN_BLAS_ROOT}/lib ENV OPENBLAS_ROOT
    PATH_SUFFIXES lib openblas-openmp openblas-pthread x86_64-linux-gnu/openblas-openmp x86_64-linux-gnu/openblas-pthread)
  if(NOT OMPNN_CBLAS_INCLUDE OR NOT OMPNN_CBLAS_LIB)
    message(FATAL_ERROR "OpenBLAS not found: set OMPNN_BLAS_ROOT to a prefix with include/cblas.h and lib/libopenblas.so")
  endif()
  target_include_directories(ompnn_backend INTERFACE ${OMPNN_CBLAS_INCLUDE})
  target_link_libraries(ompnn_backend INTERFACE ${OMPNN_CBLAS_LIB})
  set(OMPNN_BLAS_STRING "OpenBLAS ${OMPNN_CBLAS_LIB}")
elseif(OMPNN_BLAS STREQUAL "mkl")
  set(_mkl_root "${OMPNN_BLAS_ROOT}")
  if(NOT _mkl_root AND DEFINED ENV{MKLROOT})
    set(_mkl_root "$ENV{MKLROOT}")
  endif()
  find_path(OMPNN_CBLAS_INCLUDE mkl_cblas.h HINTS ${_mkl_root}/include)
  find_library(OMPNN_CBLAS_LIB NAMES mkl_rt libmkl_rt.so.3 libmkl_rt.so.2 HINTS ${_mkl_root}/lib ${_mkl_root}/lib/intel64)
  if(NOT OMPNN_CBLAS_INCLUDE OR NOT OMPNN_CBLAS_LIB)
    message(FATAL_ERROR "oneMKL not found: set OMPNN_BLAS_ROOT (or MKLROOT) to a prefix with include/mkl_cblas.h and lib/libmkl_rt.so")
  endif()
  # mkl_cblas.h has the CBLAS API; make <cblas.h> resolve to it
  file(WRITE ${CMAKE_BINARY_DIR}/mkl-cblas/cblas.h "#include <mkl_cblas.h>\n")
  target_include_directories(ompnn_backend INTERFACE ${CMAKE_BINARY_DIR}/mkl-cblas ${OMPNN_CBLAS_INCLUDE})
  target_link_libraries(ompnn_backend INTERFACE ${OMPNN_CBLAS_LIB})
  set(OMPNN_BLAS_STRING "oneMKL (mkl_rt) ${OMPNN_CBLAS_LIB}")
  message(STATUS "ompnn: mkl_rt defaults to Intel OpenMP threading; run with MKL_THREADING_LAYER=GNU under gcc/clang to share the OpenMP runtime")
else()
  message(FATAL_ERROR "OMPNN_BLAS must be openblas or mkl")
endif()
target_compile_definitions(ompnn_backend INTERFACE OMPNN_BLAS_NAME="${OMPNN_BLAS}")
if(OMPNN_GPU_BLAS)
  set(OMPNN_BLAS_STRING "${OMPNN_BLAS_STRING}; ${OMPNN_GPU_BLAS}")
endif()

# ------------------------------------------------------------------ the header-only library
add_library(ompnn INTERFACE)
add_library(ompnn::ompnn ALIAS ompnn)
target_include_directories(ompnn INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(ompnn INTERFACE ompnn_backend)

string(TOUPPER "${CMAKE_BUILD_TYPE}" _bt)
set(OMPNN_COMPILER_ID "${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION} ${CMAKE_CXX_COMPILER}")
string(REPLACE ";" " " _off "${_offload_flags}")
set(OMPNN_COMPILE_FLAGS "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${_bt}} ${OpenMP_CXX_FLAGS} ${_off}")
string(STRIP "${OMPNN_COMPILE_FLAGS}" OMPNN_COMPILE_FLAGS)
find_package(Git QUIET)
set(OMPNN_GIT_SHA "unknown")
if(GIT_FOUND)
  execute_process(COMMAND ${GIT_EXECUTABLE} -C ${CMAKE_CURRENT_SOURCE_DIR} describe --always --dirty --tags
    OUTPUT_VARIABLE OMPNN_GIT_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
endif()
configure_file(include/ompnn/build_info.hpp.in "${CMAKE_BINARY_DIR}/generated/ompnn/build_info.hpp" @ONLY)
target_include_directories(ompnn INTERFACE $<BUILD_INTERFACE:${CMAKE_BINARY_DIR}/generated>)
message(STATUS "ompnn: target=${OMPNN_TARGET} arch=${OMPNN_OFFLOAD_ARCH} blas=${OMPNN_BLAS_STRING} compiler=${OMPNN_COMPILER_ID}")

# ------------------------------------------------------------------ Python module
if(OMPNN_BUILD_PYTHON)
  find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG QUIET)
  if(NOT pybind11_FOUND)
    execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
      OUTPUT_VARIABLE _pb11_dir OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(_pb11_dir)
      list(APPEND CMAKE_PREFIX_PATH "${_pb11_dir}")
      find_package(pybind11 CONFIG QUIET)
    endif()
  endif()
  if(NOT pybind11_FOUND)
    include(FetchContent)
    FetchContent_Declare(pybind11 GIT_REPOSITORY https://github.com/pybind/pybind11.git GIT_TAG v3.1.0 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(pybind11)
  endif()
  # NO_EXTRAS: no LTO (offload toolchains and thin LTO do not mix); visibility set below
  pybind11_add_module(ompnn_python MODULE NO_EXTRAS src/python_bindings.cpp)
  target_link_libraries(ompnn_python PRIVATE ompnn)
  target_compile_definitions(ompnn_python PRIVATE OMPNN_MODULE_NAME=${OMPNN_MODULE_NAME})
  set_target_properties(ompnn_python PROPERTIES
    OUTPUT_NAME "${OMPNN_MODULE_NAME}"
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON
    BUILD_RPATH "$ORIGIN"
    INSTALL_RPATH "$ORIGIN")
  # LLVM-based compilers keep their OpenMP runtime next to the compiler (clang-18:
  # /usr/lib/llvm-18/lib, amdclang++: /opt/rocm/lib/llvm/lib): carry it as rpath
  execute_process(COMMAND ${CMAKE_CXX_COMPILER} -print-file-name=libomp.so
    OUTPUT_VARIABLE _libomp OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(_libomp AND IS_ABSOLUTE "${_libomp}" AND EXISTS "${_libomp}")
    get_filename_component(_libomp_dir "${_libomp}" DIRECTORY)
    set_target_properties(ompnn_python PROPERTIES INSTALL_RPATH "$ORIGIN;${_libomp_dir}" BUILD_RPATH "$ORIGIN;${_libomp_dir}")
    message(STATUS "ompnn: OpenMP runtime rpath ${_libomp_dir}")
  endif()
  if(SKBUILD_PROJECT_NAME)
    install(TARGETS ompnn_python DESTINATION ${SKBUILD_PROJECT_NAME})
  endif()
endif()
